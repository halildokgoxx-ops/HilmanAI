🐍 Python ile HilmanAI Entegrasyonu

HilmanAI API'sini kendi Python projelerinizde çağırmak için aşağıdaki kodu kullanabilirsiniz:

# HilmanAI Universal API Gateway
hilman_api_key = "hilman-live-8f92a1c7e45b"
model = "hilman-v1-beta"

import requests

url = "https://hilmanbey-hilmanai-v1-beta.hf.space/api/v1/chat"
headers = {
    "Content-Type": "application/json",
    "Authorization": f"Bearer {hilman_api_key}",
}
payload = {
    "model": model,
    "messages": [
        {"role": "system", "content": "Sen HilmanAI'sın."},
        {"role": "user", "content": "Merhaba! HilmanAI API ile bağlandım."}
    ],
    "temperature": 0.7,
    "max_tokens": 2048
}

response = requests.post(url, json=payload, headers=headers)
print(response.json())

⚡ Node.js / JavaScript (Fetch)

const hilman_api_key = "hilman-live-8f92a1c7e45b";
const model = "hilman-v1-beta";

async function askHilman() {
    const res = await fetch("https://hilmanbey-hilmanai-v1-beta.hf.space/api/v1/chat", {
        method: "POST",
        headers: {
            "Content-Type": "application/json",
            "Authorization": `Bearer ${hilman_api_key}`
        },
        body: JSON.stringify({
            model: model,
            messages: [{ role: "user", content: "React 19 ile modern bir dashboard örneği yaz." }]
        })
    });
    const data = await res.json();
    console.log(data.choices[0].message.content);
}

askHilman();
